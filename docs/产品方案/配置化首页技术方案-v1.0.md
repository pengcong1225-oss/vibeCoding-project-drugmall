### 1.2 核心痛点识别

#### 痛点1: 运营效率低下
- **现象**: 任何首页调整都需要开发修改代码、重新部署
- **影响**: 活动响应慢（平均3-5天），错失营销时机
- **案例**: 春季过敏活动需紧急上线，但开发排期已满

#### 痛点2: 技术债务严重
- **现象**: 单文件2100+行，职责不清晰，维护困难
- **风险**: 修改一处可能影响其他模块，Bug修复成本高
- **度量**: 新人上手理解首页逻辑需2-3天

#### 痛点3: 缺乏数据支撑
- **现象**: 无点击率、转化率等运营数据
- **后果**: 无法进行A/B测试和效果优化
- **需求**: 运营团队要求每个模块可追踪效果

#### 痛点4: 扩展性差
- **现象**: 新增模块类型需要大量代码改动
- **限制**: 业务方提出"大促专区"、"品牌馆"等新需求难以快速实现

## 二、解决方案总体设计

### 2.1 设计目标

| 目标维度 | 具体指标 | 验收标准 |
|----------|----------|----------|
| **配置化程度** | 100%元素可配置 | 所有11个模块均支持后台调整 |
| **运营效率** | 配置即时生效 | 从3-5天缩短至分钟级 |
| **代码质量** | 单文件<300行 | 拆分为20+个独立组件 |
| **数据驱动** | 全链路数据采集 | 支持点击率、曝光率统计 |
| **扩展能力** | 新模块<1天接入 | 通过配置即可添加新模块类型 |

### 2.2 核心设计理念

```
┌─────────────────────────────────────────────────────────────┐
│                    配置化首页架构                            │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  ┌──────────┐    ┌──────────┐    ┌──────────┐              │
│  │ 页面版本 │───▶│ 模块列表 │───▶│ 组件实例 │              │
│  │ (Page)   │    │(Section) │    │(Component)             │
│  └──────────┘    └──────────┘    └──────────┘              │
│       │              │               │                     │
│       ▼              ▼               ▼                     │
│  ┌──────────────────────────────────────────┐              │
│  │           数据源配置 (DataSource)          │              │
│  │  手动配置 / API接口 / 规则引擎 / 混合模式   │              │
│  └──────────────────────────────────────────┘              │
│                                                             │
│  ┌──────────────────────────────────────────┐              │
│  │           渲染引擎 (Render Engine)         │              │
│  │  动态组件加载 + 配置解析 + 数据绑定        │              │
│  └──────────────────────────────────────────┘              │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 2.3 三层配置模型

#### 第一层：页面版本层 (Page Level)
- **职责**: 管理首页的整体版本、发布状态、灰度策略
- **核心字段**: 版本号、状态（草稿/已发布/已下线）、生效时间、灰度规则
- **使用场景**: 大促前预览、灰度测试、一键回滚

#### 第二层：模块层 (Section Level)
- **职责**: 定义首页包含哪些模块及其排列顺序
- **核心字段**: 模块类型、排序权重、启用/禁用、显示条件
- **支持操作**: 启用/禁用、上下移动、复制、删除

#### 第三层：组件实例层 (Component Level)
- **职责**: 每个模块的具体配置参数和数据内容
- **核心字段**: 组件特有配置（如Banner图片URL、商品ID列表）、样式覆盖
- **灵活性**: 同一模块类型可创建多个不同配置的实例

## 四、后端API设计

### 4.1 API概览

```
/api/v1/home/
├── config/                    # 配置管理 (Admin)
│   ├── page/                  # 页面版本管理
│   │   ├── POST   create      # 创建新版本
│   │   ├── GET    list        # 版本列表
│   │   ├── GET    :id/detail  # 版本详情
│   │   ├── PUT    :id/update  # 更新版本
│   │   ├── DELETE :id         # 删除版本
│   │   └── POST  :id/publish  # 发布版本
│   ├── section/               # 模块管理
│   │   ├── POST   create      # 创建模块
│   │   ├── PUT    :id/update  # 更新模块
│   │   ├── DELETE :id         # 删除模块
│   │   └── POST   sort        # 批量排序
│   ├── component/             # 组件管理

// ============================================================
// DrugMall 首页 Store
// 职责: 管理首页配置状态、缓存、数据获取
// ============================================================

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { IHomePageVO, ISectionVO, ITabItem } from '@/types/home'
import homeApi from '@/api/home'

export const useHomeStore = defineStore('home', () => {
  // ==================== State ====================

  /** 完整的首页配置数据 */
  const homeConfig = ref<IHomePageVO | null>(null)

  /** 加载状态 */
  const loading = ref(false)

  /** 错误信息 */
  const error = ref<string | null>(null)

  /** 当前激活的Tab */
  const activeTab = ref('recommend')

  /** 最后刷新时间 */
  const lastRefreshTime = ref<number>(0)

  /** 缓存过期时间 (毫秒) 默认5分钟 */
  const CACHE_TTL = 5 * 60 * 1000

  // ==================== Getters ====================

  /** 是否已加载 */
  const isLoaded = computed(() => homeConfig.value !== null)

  /** 是否需要刷新 */
  const needsRefresh = computed(() => {
    if (!homeConfig.value) return true
    return Date.now() - lastRefreshTime.value > CACHE_TTL
  })

  /** 全局配置 */
  const globalConfig = computed(() => homeConfig.value?.globalConfig)

  /** Tab列表 */
  const tabs = computed<ITabItem[]>(() => homeConfig.value?.globalConfig?.tabs ?? [])

  /** 模块列表 (已排序) */
  const sections = computed<ISectionVO[]>(() => {
    if (!homeConfig.value?.sections) return []
    return [...homeConfig.value.sections].sort((a, b) => a.sortOrder - b.sortOrder)
  })

  /** 获取指定类型的模块 */
  const getSectionByType = (type: string) => {
    return computed(() => sections.value.find(s => s.sectionType === type))
  }

  /** 获取Tab对应的模块 */
  const getTabSections = (tabId: string) => {
    return computed(() =>
      sections.value.filter(s =>
        s.layoutConfig?.tabId === tabId
      )
    )
  }

  // ==================== Actions ====================

  /**
   * 加载首页配置
   * @param forceRefresh 是否强制刷新 (忽略缓存)
   */
  async function fetchHomeConfig(forceRefresh = false): Promise<void> {
    // 如果正在加载，直接返回
    if (loading.value) return

    // 如果不需要刷新且不是强制刷新，直接返回
    if (!forceRefresh && !needsRefresh.value && isLoaded.value) return

    loading.value = true
    error.value = null

    try {
      const response = await homeApi.getHomePageConfig()
      homeConfig.value = response.data
      lastRefreshTime.value = Date.now()

      // 如果服务端返回了缓存key，可以用于后续的缓存失效通知
      console.log(`[Home] 配置加载成功, version=${response.data.meta.version}`)
    } catch (err: any) {
      console.error('[Home] 配置加载失败:', err)
      error.value = err.message || '加载首页配置失败'

      // 如果有旧数据，继续使用旧数据 (降级策略)
      if (!homeConfig.value) {
        throw err
      }
    } finally {
      loading.value = false
    }
  }

  /**
   * 刷新单个模块数据 (用于懒加载或下拉刷新)
   */
  async function refreshSection(sectionCode: string): Promise<void> {
    try {
      const response = await homeApi.getSectionData(sectionCode)

      if (homeConfig.value?.sections) {
        const index = homeConfig.value.sections.findIndex(
          s => s.sectionCode === sectionCode
        )
        if (index !== -1) {
          homeConfig.value.sections[index] = response.data
        }
      }
    } catch (err) {
      console.error(`[Home] 刷新模块 ${sectionCode} 失败:`, err)
    }
  }

  /**
   * 切换Tab
   */
  function switchTab(tabId: string): void {
    activeTab.value = tabId
  }

  /**
   * 重置Store
   */
  function reset(): void {
    homeConfig.value = null
    loading.value = false
    error.value = null
    activeTab.value = 'recommend')
    lastRefreshTime.value = 0
  }

  /**
   * 清除缓存，下次访问会强制刷新
   */
  function invalidateCache(): void {
    lastRefreshTime.value = 0
  }

  return {
    // State
    homeConfig,
    loading,
    error,
    activeTab,

    // Getters
    isLoaded,
    needsRefresh,
    globalConfig,
    tabs,
    sections,
    getSectionByType,
    getTabSections,

    // Actions
    fetchHomeConfig,
    refreshSection,
    switchTab,
    reset,
    invalidateCache
  }
})
```

// ============================================================
// 首页动态渲染引擎
// 职责: 根据配置动态渲染模块和组件
// ============================================================

import { computed, type Component } from 'vue'
import { useHomeStore } from '@/stores/home'
import type { SectionType, ComponentType } from '@/types/home'

// 动态导入所有模块组件 (按需加载)
const sectionComponents: Record<string, () => Promise<Component>> = {
  [SectionType.SEARCH_BAR]: () => import('@/components/home/sections/SearchBarSection.vue'),
  [SectionType.TAB_NAV]: () => import('@/components/home/sections/TabNavSection.vue'),
  [SectionType.SERVICE_GRID]: () => import('@/components/home/sections/ServiceGridSection.vue'),
  [SectionType.BANNER_SUBSIDY]: () => import('@/components/home/sections/BannerSubsidySection.vue'),
  [SectionType.DOCTOR_BANNER]: () => import('@/components/home/sections/DoctorBannerSection.vue'),
  [SectionType.NEARBY_DELIVERY]: () => import('@/components/home/sections/NearbySection.vue'),
  [SectionType.WATERFALL]: () => import('@/components/home/sections/WaterfallSection.vue'),
  [SectionType.TAB_CONTENT]: () => import('@/components/home/sections/TabContentSection.vue'),
}

/**
 * 首页渲染组合函数
 *
 * @example
 * ```vue
 * <script setup>
 * import { useHomeRenderer } from '@/composables/useHomeRenderer'
 *
 * const { sortedSections, getSectionComponent, isLoading } = useHomeRenderer()
 * </script>
 *
 * <template>
 *   <div v-if="!isLoading">
 *     <component
 *       v-for="section in sortedSections"
 *       :is="getSectionComponent(section.sectionType)"
 *       :key="section.sectionCode"
 *       :section="section"
 *     />
 *   </div>
 * </template>
 * ```
 */
export function useHomeRenderer() {
  const homeStore = useHomeStore()

  /** 已排序的模块列表 */
  const sortedSections = computed(() => homeStore.sections)

  /** 是否正在加载 */
  const isLoading = computed(() => homeStore.loading)

  /**
   * 获取模块对应的Vue组件
   * @param sectionType 模块类型
   * @returns Vue组件或null
   */
  function getSectionComponent(sectionType: string): Component | null {
    const loader = sectionComponents[sectionType]
    if (!loader) {
      console.warn(`[HomeRenderer] 未知的模块类型: ${sectionType}`)
      return null
    }
    // 返回一个包装组件，用于异步加载
    return defineAsyncComponent({
      loader,
      loadingComponent: LoadingPlaceholder,
      errorComponent: ErrorPlaceholder,
      delay: 200,
      timeout: 10000
    })
  }

  /**
   * 初始化渲染器 (加载数据)
   */
  async function init(): Promise<void> {
    await homeStore.fetchHomeConfig()
  }

  /**
   * 刷新
   */
  async function refresh(): Promise<void> {
    await homeStore.fetchHomeConfig(true)
  }

  return {
    sortedSections,
    isLoading,
    getSectionComponent,
    init,
    refresh
  }
}

// 占位组件
const LoadingPlaceholder = {
  template: `<div class="section-loading"><el-skeleton :rows="3" animated /></div>`
}

const ErrorPlaceholder = {
  template: `<div class="section-error">模块加载失败</div>`
}
```
