# DrugMall 项目配置管理手册

## 概述

本手册定义DrugMall项目的所有依赖、配置项、账户凭据的统一管理规范，确保配置的安全性、可追溯性和易维护性。

---

## 1. 配置管理原则

### 1.1 核心原则

| 原则 | 说明 | 实践 |
|------|------|------|
| **安全优先** | 敏感信息加密存储 | 使用密钥管理服务，禁止硬编码 |
| **环境隔离** | 不同环境独立配置 | dev/test/staging/prod完全隔离 |
| **版本控制** | 配置变更可追溯 | 配置版本与代码版本关联 |
| **最小权限** | 按需分配访问权限 | 基于RBAC的细粒度权限控制 |
| **自动化** | 减少人工操作 | CI/CD自动注入配置 |

### 1.2 配置分级

```
┌─────────────────────────────────────────────────────────┐
│                    配置分级体系                           │
├─────────────────────────────────────────────────────────┤
│  🔴 绝密级 (Critical)                                    │
│     - 生产数据库主密码                                  │
│     - 支付网关私钥                                      │
│     - 医保对接证书                                      │
│     存储: 硬件安全模块(HSM)                             │
├─────────────────────────────────────────────────────────┤
│  🟠 机密级 (Secret)                                     │
│     - API密钥                                           │
│     - 第三方服务Token                                   │
│     - JWT签名密钥                                       │
│     存储: 密钥管理服务(KMS)                             │
├─────────────────────────────────────────────────────────┤
│  🟡 敏感级 (Sensitive)                                  │
│     - 数据库连接字符串(不含密码)                        │
│     - 内部服务地址                                      │
│     - 非生产环境配置                                    │
│     存储: 加密配置中心                                  │
├─────────────────────────────────────────────────────────┤
│  🟢 普通级 (General)                                     │
│     - 应用配置参数                                      │
│     - 功能开关                                          │
│     - 日志级别                                          │
│     存储: 配置中心                                      │
└─────────────────────────────────────────────────────────┘
```

```python
# config_validator.py
"""配置验证工具"""
import os
import re
from typing import Dict, List, Optional
from dataclasses import dataclass
from enum import Enum

class ConfigLevel(Enum):
    CRITICAL = "critical"
    SECRET = "secret"
    SENSITIVE = "sensitive"
    GENERAL = "general"

@dataclass
class ConfigItem:
    key: str
    level: ConfigLevel
    pattern: Optional[str] = None
    required: bool = True
    default: Optional[str] = None
    description: str = ""

class ConfigValidator:
    """配置验证器"""
    
    # 预定义配置项
    CONFIG_ITEMS = [
        # 数据库配置 - 绝密级
        ConfigItem("MYSQL_ROOT_PASSWORD", ConfigLevel.CRITICAL, 
                   pattern=r"^.{16,}$", 
                   description="MySQL root密码，至少16位"),
        ConfigItem("MYSQL_URL", ConfigLevel.SENSITIVE,
                   pattern=r"^jdbc:mysql://.+$",
                   description="MySQL连接URL"),
        
        # Redis配置 - 机密级
        ConfigItem("REDIS_PASSWORD", ConfigLevel.SECRET,
                   pattern=r"^.{8,}$",
                   required=False,
                   description="Redis密码"),
        
        # JWT配置 - 绝密级
        ConfigItem("JWT_SECRET", ConfigLevel.CRITICAL,
                   pattern=r"^.{32,}$",
                   description="JWT签名密钥，至少32位"),
        
        # 支付配置 - 绝密级
        ConfigItem("WECHAT_API_V3_KEY", ConfigLevel.CRITICAL,
                   description="微信支付APIv3密钥"),
        ConfigItem("WECHAT_PRIVATE_KEY_PATH", ConfigLevel.CRITICAL,
                   description="微信支付私钥路径"),
        ConfigItem("ALIPAY_PRIVATE_KEY", ConfigLevel.CRITICAL,
                   description="支付宝私钥"),
        
        # 医保配置 - 绝密级
        ConfigItem("INSURANCE_APP_SECRET", ConfigLevel.CRITICAL,
                   description="医保应用密钥"),
        ConfigItem("INSURANCE_PRIVATE_KEY", ConfigLevel.CRITICAL,
                   description="医保私钥"),
        
        # 云服务配置 - 机密级
        ConfigItem("ALIYUN_OSS_ACCESS_KEY_ID", ConfigLevel.SECRET,
                   description="阿里云OSS AccessKey ID"),
        ConfigItem("ALIYUN_OSS_ACCESS_KEY_SECRET", ConfigLevel.SECRET,
                   description="阿里云OSS AccessKey Secret"),
        
        # 短信配置 - 机密级
        ConfigItem("SMS_ALIYUN_ACCESS_KEY_ID", ConfigLevel.SECRET,
                   description="短信服务AccessKey ID"),
        ConfigItem("SMS_ALIYUN_ACCESS_KEY_SECRET", ConfigLevel.SECRET,
                   description="短信服务AccessKey Secret"),
        
        # AI/OCR配置 - 机密级
        ConfigItem("ALIYUN_OCR_ACCESS_KEY_ID", ConfigLevel.SECRET,
                   description="OCR服务AccessKey ID"),
        ConfigItem("BAIDU_OCR_API_KEY", ConfigLevel.SECRET,
                   description="百度OCR API Key"),
        
        # 应用配置 - 普通级
        ConfigItem("SERVER_PORT", ConfigLevel.GENERAL,
                   default="8080",
                   description="服务端口"),
        ConfigItem("LOG_LEVEL", ConfigLevel.GENERAL,
                   default="INFO",
                   description="日志级别"),
    ]
    
    def __init__(self):
        self.errors: List[str] = []
        self.warnings: List[str] = []
    
    def validate_all(self) -> bool:
        """验证所有配置项"""
        for item in self.CONFIG_ITEMS:
            self._validate_item(item)
        
        return len(self.errors) == 0
    
    def _validate_item(self, item: ConfigItem):
        """验证单个配置项"""
        value = os.getenv(item.key)
        
        # 检查必填项
        if item.required and not value:
            if item.default:
                self.warnings.append(
                    f"[{item.level.value}] {item.key}: 未设置，使用默认值 '{item.default}'"
                )
            else:
                self.errors.append(
                    f"[{item.level.value}] {item.key}: 必填项未设置 - {item.description}"
                )
            return
        
        # 检查值是否为空
        if not value:
            return
        
        # 验证正则表达式
        if item.pattern and not re.match(item.pattern, value):
            self.errors.append(
                f"[{item.level.value}] {item.key}: 格式不正确，期望匹配 '{item.pattern}' - {item.description}"
            )
    
    def get_report(self) -> str:
        """生成验证报告"""
        lines = [
            "=" * 80,
            "配置验证报告",
            "=" * 80,
            "",
        ]
        
        if self.errors:
            lines.extend([
                f"❌ 发现 {len(self.errors)} 个错误:",
                "-" * 80,
            ])
            for error in self.errors:
                lines.append(f"  • {error}")
            lines.append("")
        
        if self.warnings:
            lines.extend([
                f"⚠️  发现 {len(self.warnings)} 个警告:",
                "-" * 80,
            ])
            for warning in self.warnings:
                lines.append(f"  • {warning}")
            lines.append("")
        
        if not self.errors and not self.warnings:
            lines.append("✅ 所有配置验证通过！")
        
        lines.extend([
            "",
            "=" * 80,
        ])
        
        return "\n".join(lines)

# 使用示例
if __name__ == "__main__":
    validator = ConfigValidator()
    is_valid = validator.validate_all()
    print(validator.get_report())
    
    if not is_valid:
        exit(1)

```python
# config/vault_client.py
"""Vault客户端封装"""
import hvac
from typing import Optional, Dict, Any
import os

class VaultClient:
    """HashiCorp Vault客户端"""
    
    def __init__(self):
        self.client = hvac.Client(
            url=os.getenv('VAULT_ADDR', 'http://localhost:8200'),
            token=os.getenv('VAULT_TOKEN')
        )
        
        if not self.client.is_authenticated():
            raise Exception("Vault authentication failed")
    
    def get_secret(self, path: str, key: Optional[str] = None) -> Any:
        """获取密钥"""
        try:
            response = self.client.secrets.kv.v2.read_secret_version(
                path=path
            )
            data = response['data']['data']
            
            if key:
                return data.get(key)
            return data
            
        except Exception as e:
            print(f"Failed to get secret from {path}: {e}")
            return None
    
    def get_database_credentials(self, role: str = "drugmall-app") -> Dict[str, str]:
        """获取数据库动态凭证"""
        try:
            response = self.client.secrets.database.generate_credentials(
                name=role
            )
            return {
                'username': response['data']['username'],
                'password': response['data']['password']
            }
        except Exception as e:
            print(f"Failed to get database credentials: {e}")
            return {}
    
    def rotate_secret(self, path: str) -> bool:
        """轮换密钥"""
        try:
            self.client.secrets.kv.v2.rotate_secret_version(
                path=path
            )
            return True
        except Exception as e:
            print(f"Failed to rotate secret at {path}: {e}")
            return False

# 使用示例
if __name__ == "__main__":
    vault = VaultClient()
    
    # 获取应用配置
    app_config = vault.get_secret("drugmall/app-config")
    print(f"App Config: {app_config}")
    
    # 获取数据库密码
    db_password = vault.get_secret("drugmall/database", "password")
    print(f"DB Password: {'*' * len(db_password) if db_password else 'Not found'}")
    
    # 获取动态数据库凭证
    db_creds = vault.get_database_credentials()
    print(f"Dynamic DB Creds: {db_creds.get('username')}")
